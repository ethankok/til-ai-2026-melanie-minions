# Dependencies for the NeMo / Parakeet-TDT variant of the ASR container.
# Used by Dockerfile.nemo. The shipped float16 distil-whisper image keeps
# using requirements.txt; this file is a parallel build path for A/B testing.

fastapi==0.115.12
uvicorn[standard]==0.34.2

# NeMo ASR runtime. nemo_toolkit[asr] pulls in pytorch-lightning, hydra-core,
# omegaconf, and a handful of audio libs. Pin a known-good 2.x release that
# supports the Parakeet-TDT-0.6B-v2 checkpoint.
nemo_toolkit[asr]==2.0.0
# NOTE: cuda-python (for the TDT decoder's CUDA-graph fast path) is NOT
# pinned here. It used to be, but in November 2026 cuda-python 13.x landed
# with a package restructure that breaks numba 0.65.x on import:
#   ImportError: cannot import name 'cuda' from 'cuda' (unknown location)
# Pinning cuda-python<13 in this file then triggered a cascade through pip's
# resolver that downgraded torch 2.12.0 -> 2.9.1 on the Workbench host
# (because <13 forces nvidia-*-cu12 libs, but torch 2.12.0 was built against
# cu13). That broke torchvision and pytorch-lightning host-side.
# cuda-python is now installed via a dedicated `RUN pip install` line in
# Dockerfile.nemo with --no-deps, so the inference container still gets the
# CUDA-graph fast path while the host env stays untouched.
# NeMo expects a recent libsndfile and resampler. soundfile + librosa mirror
# the whisper image so audio decode behavior matches.
soundfile>=0.12.1
librosa>=0.10.0
numpy
