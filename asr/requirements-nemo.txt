# NeMo ASR runtime. nvidia/parakeet-unified-en-0.6b needs
# ConformerEncoder(att_chunk_context_size), which is present on NeMo main but
# not in the latest PyPI/tagged release (2.7.3 as of 22 May 2026). Pin the
# exact GitHub commit verified during the A/B so the image stays reproducible.
nemo_toolkit[asr] @ git+https://github.com/NVIDIA/NeMo.git@ccbbfbbdb3a4e4a4a8c06cedca6ca468e54112ac
# NOTE: cuda-python (for the TDT decoder's CUDA-graph fast path) is NOT
# pinned here. It used to be, but cuda-python 13.x has a package restructure
# that breaks numba 0.65.x on import:
#   ImportError: cannot import name 'cuda' from 'cuda' (unknown location)
# Pinning cuda-python<13 in this file then triggered a cascade through pip's
# resolver that downgraded torch 2.12.0 -> 2.9.1 on the Workbench host
# (because <13 forces nvidia-*-cu12 libs, but torch 2.12.0 was built against
# cu13). That broke torchvision and pytorch-lightning host-side.
# cuda-python is now installed via a dedicated `RUN pip install` line in the
# ASR Dockerfiles with --no-deps, so the inference container still gets the
# CUDA-graph fast path while the host env stays untouched.
# NeMo expects a recent libsndfile and resampler. soundfile + librosa mirror
# the whisper image so audio decode behavior matches.
soundfile>=0.12.1
librosa>=0.10.0
numpy
